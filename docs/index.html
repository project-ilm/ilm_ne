पूर्णाङ्क द्विआधारी_खोज(पूर्णाङ्क मानहरू[], पूर्णाङ्क गन्ती, पूर्णाङ्क साँचो) {
  पूर्णाङ्क तल्लो = 0, माथिल्लो = गन्ती - 1;
  जबसम्म (तल्लो <= माथिल्लो) {
    पूर्णाङ्क बीच = तल्लो + (माथिल्लो - तल्लो) / 2;
    यदि (मानहरू[बीच] == साँचो) फर्काऊ बीच;
    यदि (मानहरू[बीच] < साँचो) तल्लो = बीच + 1; अन्यथा माथिल्लो = बीच - 1;
  }
  फर्काऊ -1;
}
पूर्णाङ्क मुख्य(शून्य) {
  पूर्णाङ्क मानहरू[] = {2, 5, 8, 12, 16, 23, 38, 56, 72, 91};
  म_लिखो("%d %d\n", द्विआधारी_खोज(मानहरू, 10, 23), द्विआधारी_खोज(मानहरू, 10, 7));
  फर्काऊ 0;
}
